- [ ] **Step 2.1：LayerNorm 类（讲清原理 + 实现）**

markdown：
```markdown
## LayerNorm（层归一化）

把每个样本的特征归一化到均值 0、方差 1，再通过可训练的 scale/shift 还原表达力。
**作用**：稳定训练，缓解梯度爆炸/消失，让深层网络可训练。
```

code：
```python
class LayerNorm(nn.Module):
    def __init__(self, emb_dim):
        super().__init__()
        self.eps = 1e-5
        self.scale = nn.Parameter(torch.ones(emb_dim))   # 可训练缩放
        self.shift = nn.Parameter(torch.zeros(emb_dim))  # 可训练偏移

    def forward(self, x):
        mean = x.mean(dim=-1, keepdim=True)
        var = x.var(dim=-1, keepdim=True, unbiased=False)
        norm_x = (x - mean) / torch.sqrt(var + self.eps)
        return self.scale * norm_x + self.shift

# 验证
ln = LayerNorm(emb_dim=5)
out = ln(torch.ones(3, 5, dtype=torch.float32))
print("LayerNorm 输出均值≈0:", out.mean(dim=-1), "方差≈1:", out.var(dim=-1, unbiased=False))
```

```python
class GELU(nn.Module):
    def __init__(self):
        super().__init__()
    def forward(self, x):
        # GELU 的 tanh 近似公式（GPT-2 用这个）
        return 0.5 * x * (1 + torch.tanh(
            torch.sqrt(torch.tensor(2.0 / torch.pi)) *
            (x + 0.044715 * torch.pow(x, 3))
        ))

gelu, relu = GELU(), nn.ReLU()
print("GELU vs ReLU 在 [-3,3] 的曲线见笔记")
```

- [ ] **Step 3.2：FeedForward 前馈网络**

markdown：`FeedForward = Linear(emb→4×emb) → GELU → Linear(4×emb→emb)，隐藏维扩 4 倍`

```python
class FeedForward(nn.Module):
    def __init__(self, cfg):
        super().__init__()
        self.layers = nn.Sequential(
            nn.Linear(cfg["emb_dim"], 4 * cfg["emb_dim"]),  # 扩展 4 倍
            GELU(),
            nn.Linear(4 * cfg["emb_dim"], cfg["emb_dim"]),  # 压回原维
        )
    def forward(self, x):
        return self.layers(x)

ffn = FeedForward(GPT_CONFIG_124M)
x = torch.rand(2, 4, 768)
```

- [ ] **Step 4.1：残差连接（shortcut）演示**

markdown：`残差连接：x → 子层 → 子层(x) + x。让梯度能"抄近路"流回，解决深层梯度消失。`

```python
# 演示：没有残差，深层网络梯度会消失
class DeepNNNoResidual(nn.Module):
    def __init__(self, layers):
        super().__init__()
        self.layers = nn.ModuleList([nn.Linear(768, 768) for _ in range(layers)])
    def forward(self, x):
        for layer in self.layers:
            x = torch.relu(layer(x))
        return x

class DeepNNWithResidual(DeepNNNoResidual):
    def forward(self, x):
        for layer in self.layers:
            x = torch.relu(layer(x)) + x   # 残差
        return x
```

- [ ] **Step 4.2：TransformerBlock（pre-LN 结构）**

```python
from src.gpt import MultiHeadAttention  # 复用 ch03

class TransformerBlock(nn.Module):
    def __init__(self, cfg):
        super().__init__()
        self.att = MultiHeadAttention(
            d_in=cfg["emb_dim"], d_out=cfg["emb_dim"],
            context_length=cfg["context_length"],
            num_heads=cfg["n_heads"],
            dropout=cfg["drop_rate"], qkv_bias=cfg["qkv_bias"])
        self.ff = FeedForward(cfg)
        self.norm1 = LayerNorm(cfg["emb_dim"])
        self.norm2 = LayerNorm(cfg["emb_dim"])
        self.drop_shortcut = nn.Dropout(cfg["drop_rate"])

    def forward(self, x):
        # 注意力子层（pre-LN：先归一化再进子层，残差用原始 x）
        shortcut = x
        x = self.norm1(x)
        x = self.att(x)
        x = self.drop_shortcut(x)
        x = x + shortcut

        # 前馈子层
        shortcut = x
        x = self.norm2(x)
        x = self.ff(x)
        x = self.drop_shortcut(x)
        x = x + shortcut
        return x

# 验证单个 block
```

```python
def generate_text_simple(model, idx, max_new_tokens, context_size):
    model.eval()
    for _ in range(max_new_tokens):
        idx_cond = idx[:, -context_size:]   # 裁剪到上下文长度
        with torch.no_grad():
            logits = model(idx_cond)
        logits = logits[:, -1, :]           # 只取最后一个 token 的 logits
        probas = torch.softmax(logits, dim=-1)
        idx_next = torch.argmax(probas, dim=-1, keepdim=True)  # 贪婪：取最大概率
        idx = torch.cat((idx, idx_next), dim=1)
    return idx

import tiktoken
tokenizer = tiktoken.get_encoding("gpt2")
start_context = "Hello, I am"
encoded = tokenizer.encode(start_context)
idx = torch.tensor([encoded])

out_ids = generate_text_simple(model, idx, max_new_tokens=10, context_size=GPT_CONFIG_124M["context_length"])
print("生成:", tokenizer.decode(out_ids[0].tolist()))
```
